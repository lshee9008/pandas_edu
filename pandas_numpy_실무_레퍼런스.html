1. 전체 흐름

01수집ingest()
원본을 있는 그대로
02검증validate()
스키마·범위 게이트
03정제clean()
결측·중복·타입
04파생변수add_features()
행 단위 계산만
05분할train_test_split
여기서부터 누수 주의
06전처리sklearn Pipeline
스케일·인코딩·대체
07학습·평가CV → 최종 fit
08저장·추론Pipeline 통째 저장
이 구조의 핵심 규칙 3가지
  1. 04번까지는 y를 보지 않는다 — 행 하나만 보고 계산되는 파생변수만 여기서 만든다. 전체 평균·최댓값·타깃 통계가 들어가면 그 순간 데이터 누수다.
  2. 05번 이후의 모든 변환은 sklearn Pipeline 안에 넣는다 — 그래야 CV의 각 fold와 테스트셋에서 fit이 자동으로 train에만 적용된다.
  3. 저장할 때는 모델이 아니라 Pipeline을 저장한다 — 스케일러·인코더가 빠지면 그 모델은 영원히 복원 불가능하다.

3. 코드 템플릿

각 단계가 DataFrame in → DataFrame out인 순수 함수라, 어느 단계든 따로 떼어 테스트할 수 있습니다.

"""src/train.py — 전 단계를 잇는 실행 진입점"""
from __future__ import annotations
import logging, joblib, numpy as np, pandas as pd
from dataclasses import dataclass
from pathlib import Path
from sklearn.compose import ColumnTransformer
from sklearn.ensemble import RandomForestClassifier
from sklearn.impute import SimpleImputer
from sklearn.metrics import classification_report, roc_auc_score
from sklearn.model_selection import StratifiedKFold, cross_val_score, train_test_split
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import OneHotEncoder, StandardScaler

log = logging.getLogger(__name__)
logging.basicConfig(level=logging.INFO, format='%(levelname)s | %(message)s')

# ---------- 설정: 코드 안에 상수를 흩뿌리지 않는다 ----------
@dataclass(frozen=True)
class Config:
    target:    str   = 'survived'
    test_size: float = 0.2
    seed:      int   = 42
    num_cols:  tuple = ('age', 'fare', 'sibsp')
    cat_cols:  tuple = ('sex', 'pclass', 'embarked')
    out_dir:   Path  = Path('models')

# ---------- 01. 수집: 원본을 그대로 읽기만 ----------
def ingest(cfg) -> pd.DataFrame:
    return pd.read_csv('data/raw/train.csv')

# ---------- 02. 검증: 여기서 막히면 아래로 못 내려간다 ----------
def validate(df, cfg) -> pd.DataFrame:
    required = set(cfg.num_cols) | set(cfg.cat_cols) | {cfg.target}
    missing  = required - set(df.columns)
    assert not missing,               f'컬럼 누락: {missing}'
    assert len(df) > 0,               '빈 데이터'
    assert df[cfg.target].notna().any(), '타깃 전체 결측'
    log.info(f'검증 통과 | shape={df.shape} 결측률={df.isna().mean().mean():.1%}')
    return df

# ---------- 03. 정제: 되돌릴 수 있는 변환만 ----------
def clean(df, cfg) -> pd.DataFrame:
    return (df
            .dropna(subset=[cfg.target])
            .drop_duplicates()
            .assign(**{cfg.target: lambda d: d[cfg.target].astype(int)})
            .reset_index(drop=True))

# ---------- 04. 파생변수: 행 하나만 보고 계산되는 것만 ----------
def add_features(df) -> pd.DataFrame:
    return df.assign(
        family_size = lambda d: d.sibsp.fillna(0) + 1,
        is_alone    = lambda d: (d.sibsp.fillna(0) == 0).astype(int),
        log_fare    = lambda d: np.log1p(d.fare),
    )
    # 금지 예: d.fare / d.fare.mean()  ← 전체 평균은 누수

# ---------- 06. 전처리 Pipeline: 컬럼 종류별로 갈라서 처리 ----------
def build_pipeline(cfg) -> Pipeline:
    num = Pipeline([('imp', SimpleImputer(strategy='median')),
                    ('sc',  StandardScaler())])
    cat = Pipeline([('imp', SimpleImputer(strategy='most_frequent')),
                    ('oh',  OneHotEncoder(handle_unknown='ignore', drop='if_binary'))])
    pre = ColumnTransformer(
        [('num', num, list(cfg.num_cols) + ['family_size', 'log_fare']),
         ('cat', cat, list(cfg.cat_cols))],
        remainder='drop')                # 명시 안 한 컬럼은 버림
    return Pipeline([('pre',   pre),
                     ('model', RandomForestClassifier(n_estimators=200,
                                                      random_state=cfg.seed))])

# ---------- 05·07·08. 분할 → 평가 → 저장 ----------
def run(cfg = Config()):
    df = add_features(clean(validate(ingest(cfg), cfg), cfg))
    X, y = df.drop(columns=cfg.target), df[cfg.target]

    Xtr, Xte, ytr, yte = train_test_split(
        X, y, test_size=cfg.test_size, stratify=y, random_state=cfg.seed)

    pipe = build_pipeline(cfg)

    # CV: fold마다 전처리를 train에만 fit → 누수 원천 차단
    cv = cross_val_score(pipe, Xtr, ytr, scoring='roc_auc',
                         cv=StratifiedKFold(5, shuffle=True, random_state=cfg.seed))
    log.info(f'CV ROC-AUC = {cv.mean():.4f} ± {cv.std():.4f}')

    pipe.fit(Xtr, ytr)
    proba = pipe.predict_proba(Xte)[:, 1]
    log.info(f'Test ROC-AUC = {roc_auc_score(yte, proba):.4f}')
    print(classification_report(yte, pipe.predict(Xte), digits=4))

    # 모델만이 아니라 설정까지 함께 저장 → 나중에 재현 가능
    cfg.out_dir.mkdir(parents=True, exist_ok=True)
    joblib.dump({'pipeline': pipe, 'config': cfg, 'columns': list(X.columns)},
                cfg.out_dir / 'model.joblib')

if __name__ == '__main__':
    run()
"""src/predict.py — 저장한 Pipeline은 원본 스키마를 그대로 받는다"""
import joblib, pandas as pd

art  = joblib.load('models/model.joblib')
pipe = art['pipeline']

new = pd.read_csv('data/raw/new.csv')          # 전처리 안 된 원본 그대로
new = add_features(clean(validate(new, art['config']), art['config']))

pred  = pipe.predict(new)                       # 스케일·인코딩 자동 적용
proba = pipe.predict_proba(new)[:, 1]

4. 단계별 원칙

단계해야 할 일하면 안 되는 일검증 방법
01 수집원본을 그대로 읽고 data/raw/는 읽기 전용 취급읽으면서 컬럼 삭제·이름 변경df.shape, df.dtypes 로그
02 검증assert로 컬럼·행수·타깃 결측 확인조용히 넘어가기 — 여기서 못 잡으면 끝까지 간다일부러 컬럼 지우고 실패하는지 확인
03 정제중복 제거, 타입 변환, 타깃 결측 행 제거피처 결측을 여기서 채우기 (평균이 전체에서 계산됨)정제 전후 행수 로그
04 파생행 단위 계산 (비율, 로그, 날짜 분해, 문자열 길이)mean()·max()·타깃 인코딩 등 전체 통계한 행만 넣어도 같은 값이 나오는지 확인
05 분할stratify=y, random_state 고정. 시계열이면 시간 순 분할분할 전에 스케일링·결측 대체y_train.value_counts(normalize=True)
06 전처리ColumnTransformer로 수치/범주 분리 처리테스트셋에 fit 또는 fit_transformCV 점수와 테스트 점수 격차 확인
07 평가CV로 안정성 확인, 문제에 맞는 지표 선택단일 split 정확도만 보고 결론seed를 바꿔 재실행
08 저장Pipeline + 설정 + 컬럼 목록을 함께 저장모델 객체만 저장불러와서 원본 스키마로 predict

5. 지표 선택 (07단계에서 자주 틀리는 부분)

상황쓸 지표이유
클래스 균형 분류accuracy, f1기본값으로 무난
불균형 분류 (사기·이상탐지)average_precision, f1, PR curve정확도는 "전부 정상"만 찍어도 99% 나옴
놓치면 안 되는 문제 (암 진단)recall (양성 클래스), 혼동행렬거짓음성 1건의 비용이 압도적
순위·확률이 필요roc_auc, brier_score_loss임계값에 무관한 판별력을 본다
회귀 일반MAE, RMSEMAE는 이상치에 둔감, RMSE는 큰 오차를 벌준다
회귀 · 값의 크기가 제각각MAPE, 로그 변환 후 RMSE절대오차로는 큰 값이 지표를 지배

6. 자주 나오는 안티패턴

❌ 분할 전에 스케일링 scaler.fit(X) 후 split — 테스트셋의 평균·최댓값이 학습에 새어 들어갑니다.

✅ Pipeline([('sc',StandardScaler()),('m',model)]) 로 묶고 cross_val_score
❌ 모델만 저장 joblib.dump(model) — 스케일러·인코더가 빠져 복원 불가.

✅ joblib.dump(pipeline)
❌ iterrows로 행 반복 10만 행에서 수십 초. 파이썬 루프는 벡터화보다 100배 이상 느립니다.

✅ np.where / np.select / .str / .dt 접근자
❌ 반복문 안에서 concat 매번 전체를 새로 복사해 O(n²).

✅ 리스트에 모았다가 pd.concat(chunks) 한 번
❌ 연쇄 인덱싱 대입 df[df.a>0]['b'] = 1 — 사본에 써서 반영 안 됩니다.

✅ df.loc[df.a>0, 'b'] = 1
❌ 단일 split 결과로 모델 비교 테스트 100건이면 샘플 2개 차이가 2%p입니다.

✅ cross_val_score 평균 ± 표준편차로 비교
제출 전 최종 체크리스트 ☐ random_state를 split·모델·CV 전부에 고정했는가
☐ 테스트셋에 fit이 단 한 번도 호출되지 않았는가
☐ 파생변수에 전체 통계(mean, max, 타깃값)가 섞이지 않았는가
☐ 저장한 파일을 불러와 원본 스키마로 predict가 되는가
☐ CV 점수와 테스트 점수 격차가 CV 표준편차 안에 들어오는가
☐ 지표가 문제 성격에 맞는가 (불균형인데 accuracy를 보고 있지는 않은가)
